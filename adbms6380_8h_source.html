31// PEC size in bytes for ADBMS6380 commands and data packets. Is 2 bytes for both PEC10 (data) and PEC15 (commands).
32static constexpr size_t ADBMS6380_PEC_SIZE = 2U;
33// Raw command size (bytes) for one command without PEC.
34static constexpr size_t ADBMS6380_COMMAND_RAW_SIZE = 2U;
35// Command packet size (bytes) including PEC.
36static constexpr size_t ADBMS6380_COMMAND_PKT_SIZE = ADBMS6380_COMMAND_RAW_SIZE + ADBMS6380_PEC_SIZE;
37// Single data payload size (bytes) for one chip without PEC.
38static constexpr size_t ADBMS6380_SINGLE_DATA_RAW_SIZE = 6U;
39// Single data packet size (bytes) for one chip including PEC.
40static constexpr size_t ADBMS6380_SINGLE_DATA_PKT_SIZE =
41 ADBMS6380_SINGLE_DATA_RAW_SIZE + ADBMS6380_PEC_SIZE;
42
43// Number of commands to read all 16 cell voltages (5 for 3 cells each, 1 for the last cell)
44// {RDCVA, RDCVB, RDCVC, RDCVD, RDCVE, RDCVF}
45static constexpr size_t ADBMS6380_RDCV_CMD_COUNT = 6U;
46
47// Number of commands to read all 10 GPIO voltages (3 for 3 GPIOs each, 1 for the last GPIO)
48// {RDAUXA, RDAUXB, RDAUXC, RDAUXD}
49static constexpr size_t ADBMS6380_RDAUX_CMD_COUNT = 4U;
50
51// Max SPI TX is a command + all the data packets for all the chips in the daisy chain.
52static constexpr size_t ADBMS6380_SPI_TX_BUFFER_SIZE =
53 ADBMS6380_COMMAND_PKT_SIZE + (ADBMS_CHIP_COUNT * ADBMS6380_SINGLE_DATA_PKT_SIZE);
54// Max SPI RX is all the data packets for all the chips in the daisy chain.
55static constexpr size_t ADBMS6380_SPI_RX_BUFFER_SIZE =
56 ADBMS6380_SINGLE_DATA_PKT_SIZE * ADBMS_CHIP_COUNT;
