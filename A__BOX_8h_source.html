342 memset((void *)&can_data, 0, sizeof(can_data));
343 // Underflow the last_rx timers to ensure messages are stale until first rx
344 // Setup the stale function pointers
345 can_data.elcon_status.last_rx = (uint32_t) -(ELCON_STATUS_STALE_TIMEOUT_MS + 1);
346 can_data.elcon_status.is_stale = is_elcon_status_stale;
347 can_data.charge_request.last_rx = (uint32_t) -(CHARGE_REQUEST_STALE_TIMEOUT_MS + 1);
348 can_data.charge_request.is_stale = is_charge_request_stale;
349 can_data.dashboard_fault_sync.last_rx = (uint32_t) -(DASHBOARD_FAULT_SYNC_STALE_TIMEOUT_MS + 1);
350 can_data.dashboard_fault_sync.is_stale = is_dashboard_fault_sync_stale;
351 can_data.main_module_fault_sync.last_rx = (uint32_t) -(MAIN_MODULE_FAULT_SYNC_STALE_TIMEOUT_MS + 1);

353 can_data.pdu_fault_sync.last_rx = (uint32_t) -(PDU_FAULT_SYNC_STALE_TIMEOUT_MS + 1);

357}
