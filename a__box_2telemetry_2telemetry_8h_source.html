14static inline float vbatt_to_voltage(uint16_t vbatt_raw) {
15 // Example conversion:
16 // 600V actual => linearly scaled to 2v -> linearly scaled to 4v -> (4.7/5.7) divider to ~3.3v
17 // Full conversion:
18 // V * (1/300) * 2 * (4.7/5.7) * 4095/3.3 = ADC
19 // V = (300) * (1/2) * (5.7/4.7) * (3.3/4095) * ADC
20
21 static constexpr float ADC_TO_PACK_VOLTAGE = 300.0f * (1.0f / 2.0f) * (5.7f / 4.7f) * (3.3f / 4095.0f);
22
23 float voltage = vbatt_raw * ADC_TO_PACK_VOLTAGE;
24
25 return voltage;
26}

29static inline float isense_to_current(uint16_t isense_raw) {
30 static constexpr float ADC_VREF = 3.3f;
31 static constexpr float ADC_MAX = 4095.0f;
32 static constexpr float ADC_TO_VOLTS = ADC_VREF / ADC_MAX;
33
34 static constexpr float DIV_R1 = 2400.0f;
35 static constexpr float DIV_R2 = 4700.0f;
36 static constexpr float DIV_GAIN = (DIV_R1 + DIV_R2) / DIV_R2;
37
38 static constexpr float V_OFFSET = 2.5f;
39 static constexpr float G = 10.0e-3f;
40
41 float v_adc = isense_raw * ADC_TO_VOLTS;
42 float v_sensor = v_adc * DIV_GAIN;
43 float current = (v_sensor - V_OFFSET) / G; // data
44
45 // Apply offset correction
46 static constexpr float ISENSE_OFFSET_AMPS = 2.5f; // todo tune this
47 current += ISENSE_OFFSET_AMPS;
48
49 return current;
50}
